🔄 FLUJO IMPLEMENTADO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PASO 1: BUSCAR TICKET
  Input:  Nombre empresa, folio, monto
  Output: Detalles del ticket si existe
  BD:     SELECT FROM tickets
  
PASO 2: VALIDAR DATOS
  Validaciones:
  ✅ Email válido (filter_var)
  ✅ RFC 12-13 caracteres
  ✅ CP 5 dígitos
  ✅ Tipo persona = Física/Moral
  ✅ Datos requeridos presentes
  
PASO 3: CREAR/REUTILIZAR USUARIO
  BD:     INSERT/SELECT usuarios (tipo_cliente='invitado')
  Nota:   Sin password requerida
  
PASO 4: GUARDAR DATOS FISCALES
  BD:     INSERT/UPDATE datos_fiscales_usuario
  Campos: RFC, razón social, régimen, CP, tipo persona, domicilio
  
PASO 5: GENERAR FOLIO
  Lógica:  SELECT MAX(folio) + 1
  Formato: Serie + 6 dígitos (Ej: A000001)
  
PASO 6: CREAR FACTURA EN BD
  BD:     INSERT facturas (folio, RFC, totales, etc)
  Estado: 'pendiente'
  
PASO 7: COPIAR DETALLES DEL TICKET
  BD:     SELECT ticket_detalle → INSERT facturas_detalles x N
  Cálculo: IVA automático (16%)
  
PASO 8: MARCAR TICKET COMO FACTURADO
  BD:     UPDATE tickets SET estatus='facturado', id_factura=?
  Link:   Vincular ticket ↔ factura
  
PASO 9: GENERAR XML
  API:    POST core/generar-xml.php
  Output: XML CFDI válido
  BD:     UPDATE facturas SET xml_path=?
  
PASO 10: TIMBRAR CON SAT
  API:    POST core/timbrar-xml.php
  Service: Finkok (HTTP)
  Output: UUID del timbrado
  BD:     UPDATE facturas SET uuid=?, estatus='timbrado'
  
PASO 11: GENERAR PDF
  Service: mPDF (FacturaPdfService)
  Output:  PDF con factura formateada
  BD:     UPDATE facturas SET pdf_path=?
  Nota:   No bloquea si falla (warning en logs)
  
PASO 12: ENVIAR EMAIL
  Service: PHPMailer + SMTP
  To:      Correo del cliente
  Subject: Factura Electrónica {folio} - {empresa}
  Adjuntos: PDF + XML
  Nota:   No bloquea si falla (warning en logs)
  
PASO 13: RESPUESTA AL CLIENTE
  Response: JSON con confirmación
  Status:   HTTP 200 (éxito) / 400 (error)
  Data:     id_factura, folio, uuid, correo

TABLAS IMPACTADAS:
  
  usuarios
    ├─ INSERT: Nuevo usuario (tipo_cliente='invitado')
    └─ Verificación: usuario existente por correo
  
  datos_fiscales_usuario
    ├─ INSERT: Nuevos datos fiscales
    └─ UPDATE: Actualizar datos existentes
  
  facturas
    ├─ INSERT: Nueva factura con folio
    ├─ UPDATE: xml_path, uuid, pdf_path, estatus
    └─ Campos nuevos: estatus='timbrado'
  
  facturas_detalles
    ├─ INSERT: Copiar detalles del ticket
    └─ Líneas x N (según ticket_detalle)
  
  tickets
    └─ UPDATE: estatus='facturado', id_factura link
