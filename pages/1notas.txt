<script>
    document.addEventListener('DOMContentLoaded', function() {
        // Cargar sucursales
        cargarSucursales();
        
        // Establecer fecha actual
        document.getElementById('fechaCompra').valueAsDate = new Date();
        
        // Event listeners
        document.getElementById('cpFiscal').addEventListener('blur', buscarColoniasPorCP);
        document.getElementById('habilitarTerceros').addEventListener('change', toggleDatosTerceros);
        document.getElementById('rfcReceptor').addEventListener('input', formatearRFC);
    });

    function formatearRFC(e) {
        e.target.value = e.target.value.toUpperCase().replace(/[^A-Z0-9&]/g, '');
    }

    async function cargarSucursales() {
        try {
            const response = await fetch('core/consultar-tickets.php?obtener_sucursales=1');
            const data = await response.json();
            if (data.success && data.sucursales) {
                const select = document.getElementById('sucursalReceptor');
                data.sucursales.forEach(s => {
                    select.innerHTML += `<option value="${s.id}">${s.nombre}</option>`;
                });
            }
        } catch (error) {
            console.error('Error cargando sucursales:', error);
        }
    }

    async function buscarColoniasPorCP() {
        const cp = document.getElementById('cpFiscal').value;
        if (cp.length !== 5) return;
        
        try {
            const response = await fetch(`core/obtener-colonias-cp.php?cp=${cp}`);
            const data = await response.json();
            
            if (data.success) {
                const selectColonia = document.getElementById('coloniaFiscal');
                selectColonia.innerHTML = '<option value="">Seleccionar colonia...</option>';
                
                data.colonias.forEach(colonia => {
                    selectColonia.innerHTML += `<option value="${colonia.nombre}">${colonia.nombre}</option>`;
                });
                
                document.getElementById('municipioFiscal').value = data.municipio || '';
                document.getElementById('estadoFiscal').value = data.estado || '';
            }
        } catch (error) {
            console.error('Error buscando colonias:', error);
        }
    }

    function toggleDatosTerceros() {
        const checked = document.getElementById('habilitarTerceros').checked;
        document.getElementById('datosTerceros').classList.toggle('d-none', !checked);
    }

    function formatMoney(amount) {
        return new Intl.NumberFormat('es-MX', { style: 'currency', currency: 'MXN' }).format(amount || 0);
    }

    async function buscarTicket() {
        const folio = document.getElementById('folioTicket').value.trim();
        if (!folio) {
            Swal.fire('Error', 'Ingresa el número de folio del ticket', 'warning');
            return;
        }

        Swal.fire({
            title: 'Buscando ticket...',
            allowOutsideClick: false,
            didOpen: () => Swal.showLoading()
        });

        try {
            const response = await fetch(`core/consultar-tickets.php?folio=${folio}`);
            const data = await response.json();
            
            Swal.close();
            
            if (data.success && data.ticket) {
                cargarDatosTicket(data.ticket);
                Swal.fire('Éxito', 'Ticket encontrado correctamente', 'success');
            } else {
                Swal.fire('No encontrado', data.message || 'No se encontró el ticket', 'error');
            }
        } catch (error) {
            Swal.close();
            console.error('Error:', error);
            Swal.fire('Error', 'No se pudo conectar con el servidor', 'error');
        }
    }

    function cargarDatosTicket(ticket) {
        // Cargar datos del ticket
        document.getElementById('fechaCompra').value = ticket.fecha_venta?.split(' ')[0] || '';
        
        // Cargar productos
        const tbody = document.getElementById('tbodyProductos');
        if (ticket.productos && ticket.productos.length > 0) {
            tbody.innerHTML = ticket.productos.map(p => `
                <tr>
                    <td>${p.descripcion || p.nombre}</td>
                    <td class="text-center">${p.cantidad}</td>
                    <td class="text-end">${formatMoney(p.precio_unitario)}</td>
                    <td class="text-end">${formatMoney(p.importe)}</td>
                </tr>
            `).join('');
        }
        
        // Actualizar totales
        document.getElementById('subtotalCompra').textContent = formatMoney(ticket.subtotal);
        document.getElementById('ivaCompra').textContent = formatMoney(ticket.impuesto_t);
        document.getElementById('totalCompra').textContent = formatMoney(ticket.importe_t);
    }

    function previsualizarFactura() {
        const datos = recopilarDatosFactura();
        
        if (!validarDatosRequeridos(datos)) return;

        Swal.fire({
            title: 'Vista Previa de Factura',
            html: generarHTMLPreview(datos),
            width: '80%',
            showCloseButton: true,
            confirmButtonText: 'Cerrar',
            customClass: { popup: 'swal-wide' }
        });
    }

    function validarDatosRequeridos(datos) {
        const camposRequeridos = [
            { campo: 'rfcReceptor', nombre: 'RFC' },
            { campo: 'nombreReceptor', nombre: 'Nombre o Razón Social' },
            { campo: 'usoCfdi', nombre: 'Uso de CFDI' },
            { campo: 'regimenFiscal', nombre: 'Régimen Fiscal' },
            { campo: 'cpFiscal', nombre: 'Código Postal' },
            { campo: 'correosReceptor', nombre: 'Correo Electrónico' }
        ];

        for (const item of camposRequeridos) {
            if (!datos[item.campo]) {
                Swal.fire('Campo requerido', `El campo "${item.nombre}" es obligatorio`, 'warning');
                document.getElementById(item.campo).focus();
                return false;
            }
        }

        // Validar RFC
        if (datos.rfcReceptor.length < 12 || datos.rfcReceptor.length > 13) {
            Swal.fire('RFC inválido', 'El RFC debe tener 12 o 13 caracteres', 'warning');
            return false;
        }

        // Validar correos
        const correos = datos.correosReceptor.split(',').map(c => c.trim());
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        for (const correo of correos) {
            if (!emailRegex.test(correo)) {
                Swal.fire('Correo inválido', `El correo "${correo}" no es válido`, 'warning');
                return false;
            }
        }

        return true;
    }

    function generarFactura() {
        const datos = recopilarDatosFactura();
        
        if (!validarDatosRequeridos(datos)) return;

        Swal.fire({
            title: '¿Generar Factura?',
            text: 'Se generará la factura CFDI con los datos proporcionados',
            icon: 'question',
            showCancelButton: true,
            confirmButtonColor: '#198754',
            confirmButtonText: 'Sí, generar',
            cancelButtonText: 'Cancelar'
        }).then((result) => {
            if (result.isConfirmed) {
                procesarGeneracionFactura(datos);
            }
        });
    }

    async function procesarGeneracionFactura(datos) {
        Swal.fire({
            title: 'Generando factura...',
            html: '<p>Por favor espera mientras se procesa tu factura CFDI</p>',
            allowOutsideClick: false,
            didOpen: () => Swal.showLoading()
        });

        try {
            const response = await fetch('core/generar-factura-cfdi.php', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(datos)
            });
            
            const data = await response.json();
            Swal.close();
            
            if (data.success) {
                Swal.fire({
                    icon: 'success',
                    title: '¡Factura Generada!',
                    html: `
                        <p>Tu factura ha sido generada correctamente.</p>
                        <p><strong>UUID:</strong> ${data.uuid || 'N/A'}</p>
                        <p>Se enviará al correo: ${datos.correosReceptor}</p>
                    `,
                    confirmButtonText: 'Descargar PDF',
                    showCancelButton: true,
                    cancelButtonText: 'Cerrar'
                }).then((result) => {
                    if (result.isConfirmed && data.pdf_url) {
                        window.open(data.pdf_url, '_blank');
                    }
                });
            } else {
                Swal.fire('Error', data.message || 'Error al generar la factura', 'error');
            }
        } catch (error) {
            Swal.close();
            console.error('Error:', error);
            Swal.fire('Error', 'No se pudo conectar con el servidor', 'error');
        }
    }
