<script>
    document.addEventListener('DOMContentLoaded', function() {
        // Elementos del DOM
        const form = document.getElementById('formBuscarTicket');
        const numeroTicketInput = document.getElementById('numeroTicket');
        const montoTotalInput = document.getElementById('montoTotal');
        const fechaCompraInput = document.getElementById('fechaCompra');
        const lugarCompraInput = document.getElementById('lugarCompraInput');
        const usoCfdiSelect = document.getElementById('usoCfdi');
        
        // Establecer fecha máxima (hoy)
        fechaCompraInput.max = new Date().toISOString().split('T')[0];
        
        // Validación en tiempo real del folio
        numeroTicketInput.addEventListener('input', function() {
            const value = this.value;
            if (value.length > 0) {
                this.classList.add('is-valid');
                this.classList.remove('is-invalid');
            } else {
                this.classList.remove('is-valid', 'is-invalid');
            }
        });
        
        // Validación del monto
        montoTotalInput.addEventListener('input', function() {
            const value = parseFloat(this.value);
            if (value > 0) {
                this.classList.add('is-valid');
                this.classList.remove('is-invalid');
            } else if (this.value !== '') {
                this.classList.add('is-invalid');
                this.classList.remove('is-valid');
            } else {
                this.classList.remove('is-valid', 'is-invalid');
            }
        });
        
        // Validación de la fecha
        fechaCompraInput.addEventListener('change', function() {
            const selectedDate = new Date(this.value);
            const today = new Date();
            
            if (selectedDate <= today) {
                this.classList.add('is-valid');
                this.classList.remove('is-invalid');
            } else {
                this.classList.add('is-invalid');
                this.classList.remove('is-valid');
            }
        });
        
        // Envío del formulario
        form.addEventListener('submit', function(e) {
            e.preventDefault();
            
            if (!this.checkValidity()) {
                this.reportValidity();
                return;
            }
            
            // Simulación de procesamiento
            const btnSubmit = document.getElementById('btnGenerarFactura');
            const originalText = btnSubmit.innerHTML;
            
            btnSubmit.disabled = true;
            btnSubmit.innerHTML = '<i class="bi bi-hourglass-split me-2"></i>Generando factura...';
            
            // Simular proceso de generación
            setTimeout(() => {
                btnSubmit.innerHTML = '<i class="bi bi-check-circle-fill me-2"></i>¡Factura generada!';
                
                setTimeout(() => {
                    alert('¡Factura generada exitosamente!\n\nSe ha enviado a tu correo electrónico registrado.');
                    
                    // Resetear formulario
                    form.reset();
                    document.querySelectorAll('.is-valid, .is-invalid').forEach(el => {
                        el.classList.remove('is-valid', 'is-invalid');
                    });
                    
                    btnSubmit.disabled = false;
                    btnSubmit.innerHTML = originalText;
                }, 1500);
            }, 2000);
        });
    });
    
    // Filtrar sucursales
    document.getElementById('busquedaSucursal').addEventListener('input', function() {
        const filtro = this.value.toLowerCase();
        const items = document.querySelectorAll('.sucursal-card');
        
        items.forEach(function(item) {
            const texto = item.textContent.toLowerCase();
            const container = item.closest('.col-md-6');
            
            if (texto.includes(filtro)) {
                container.style.display = '';
            } else {
                container.style.display = 'none';
            }
        });
    });
    
    // Seleccionar sucursal y cerrar modal
    function seleccionarSucursal(nombre) {
        document.getElementById('lugarCompraInput').value = nombre;
        const modal = bootstrap.Modal.getOrCreateInstance(document.getElementById('modalSucursales'));
        modal.hide();
        
        // Agregar feedback visual
        const input = document.getElementById('lugarCompraInput');
        input.classList.add('is-valid');
        input.classList.remove('is-invalid');
    }
</script>
